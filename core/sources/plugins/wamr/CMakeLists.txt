if(WIN32)
  set(WAMR_BUILD_PLATFORM windows)
elseif(ANDROID)
  set(WAMR_BUILD_PLATFORM android)
elseif(APPLE)
  set(WAMR_BUILD_PLATFORM darwin)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(WAMR_BUILD_PLATFORM linux)
else()
  message(FATAL_ERROR "The Mobagen WAMR backend does not support '${CMAKE_SYSTEM_NAME}' yet")
endif()

if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|ARM64|aarch64|AARCH64)")
  set(WAMR_BUILD_TARGET AARCH64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(WAMR_BUILD_TARGET X86_64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
  set(WAMR_BUILD_TARGET X86_32)
else()
  message(FATAL_ERROR "The Mobagen WAMR backend requires a 32-bit or 64-bit target")
endif()

set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_MULTI_MODULE 0)
set(WAMR_BUILD_THREAD_MGR 0)
set(WAMR_BUILD_LIB_PTHREAD 0)
set(WAMR_BUILD_SHARED_MEMORY 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
set(WAMR_BUILD_BULK_MEMORY 1)
set(WAMR_BUILD_MEMORY64 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)

set(WAMR_ROOT_DIR "${wasm-micro-runtime_SOURCE_DIR}")
include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")

add_library(mobagen_wamr_vm STATIC ${WAMR_RUNTIME_LIB_SOURCE})
add_library(mobagen::wamr_vm ALIAS mobagen_wamr_vm)
target_compile_features(mobagen_wamr_vm PUBLIC c_std_11)
target_include_directories(mobagen_wamr_vm PUBLIC "${WAMR_ROOT_DIR}/core/iwasm/include")
set_target_properties(mobagen_wamr_vm PROPERTIES POSITION_INDEPENDENT_CODE ON)

find_package(Threads REQUIRED)
target_link_libraries(mobagen_wamr_vm PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(NOT MSVC)
  target_link_libraries(mobagen_wamr_vm PRIVATE m)
endif()
